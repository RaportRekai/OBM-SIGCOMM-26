# ==============================================================================
# OBM HARDWARE SIMULATION SWEEP
# Generates:
#   FINAL_ALL_RESULTS_UNIFORM.txt
#   FINAL_ALL_RESULTS_ZIPF.txt
#   FINAL_ALL_RESULTS_SKEWED.txt
# ==============================================================================

# 1. SETUP
cd [get_property DIRECTORY [current_project]]

set dist_list  {0 1 2}
set dist_names {UNIFORM ZIPF SKEWED}

# OBM burst percentages
set burst_list {10 20 30 40 50}

set verilog_filename "VERILOG_RESULTS.txt"

# 2. RUN EACH DISTRIBUTION
for {set d 0} {$d < [llength $dist_list]} {incr d} {

    set dist      [lindex $dist_list $d]
    set dist_name [lindex $dist_names $d]

    set final_output "FINAL_ALL_RESULTS_${dist_name}.txt"

    # Clear/create the output file
    set f [open $final_output "w"]
    puts $f "Simulation Sweep Results - [clock format [clock seconds]]"
    puts $f "Distribution: $dist_name"
    puts $f "=================================================="
    close $f

    puts ""
    puts "=================================================="
    puts "OBM DISTRIBUTION: $dist_name"
    puts "RESULTS: [file normalize $final_output]"
    puts "=================================================="

    # Run all bursts
    foreach burst $burst_list {

        puts "RUNNING: Dist=$dist ($dist_name) | Burst=$burst%"

        # Set testbench parameters
        set_property generic \
            "DISTRIBUTION_TYPE=$dist BURST_PERCENT=$burst" \
            [get_filesets sim_1]

        reset_simulation -simset sim_1 -mode behavioral
        launch_simulation
        run -all
        close_sim -force

        # Find VERILOG_RESULTS.txt
        set generated_files \
            [glob -nocomplain ./*.sim/sim_1/behav/xsim/$verilog_filename]

        if {$generated_files == ""} {
            set generated_files \
                [glob -nocomplain ./*/*.sim/sim_1/behav/xsim/$verilog_filename]
        }

        if {$generated_files != ""} {

            set source_file [lindex $generated_files 0]

            if {![catch {exec cat $source_file} content]} {

                set fout [open $final_output "a"]
                puts $fout $content
                close $fout

                puts "  -> Saved to $final_output"

            } else {
                puts "  ERROR: Could not read $source_file"
            }

        } else {
            puts "  ERROR: VERILOG_RESULTS.txt not found."
        }
    }

    puts "COMPLETED: $dist_name"
}

puts ""
puts "----------------------------------------------------------------"
puts "OBM SWEEP COMPLETE"
puts "Generated:"
puts "  FINAL_ALL_RESULTS_UNIFORM.txt"
puts "  FINAL_ALL_RESULTS_ZIPF.txt"
puts "  FINAL_ALL_RESULTS_SKEWED.txt"
puts "----------------------------------------------------------------"
