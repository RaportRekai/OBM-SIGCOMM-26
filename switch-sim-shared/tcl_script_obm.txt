# ==============================================================================
#  SAFE APPEND SWEEP (Saves data before Vivado wipes it)
# ==============================================================================

# 1. SETUP
cd [get_property DIRECTORY [current_project]]

set dist_list {0 1 2}
set dist_names {UNIFORM ZIPF SKEWED}

set burst_list {10 20 30 40 50}

# The file Verilog writes to (inside the sim folder)
set verilog_filename "VERILOG_RESULTS.txt"

# 2. CLEAR THE FINAL OUTPUT FILES
foreach dist_name $dist_names {

    set final_output "FINAL_ALL_RESULTS_${dist_name}.txt"

    set f [open $final_output "w"]
    puts $f "Simulation Sweep Results - [clock format [clock seconds]]"
    puts $f "=================================================="
    close $f

    puts "RESULTS WILL BE SAVED TO: [file normalize $final_output]"
}

# 3. RUN THE SWEEP
foreach dist $dist_list {

    # Select output file corresponding to this distribution
    set dist_name [lindex $dist_names $dist]
    set final_output "FINAL_ALL_RESULTS_${dist_name}.txt"

    foreach burst $burst_list {

        puts "RUNNING: Dist=$dist ($dist_name) | Burst=$burst%"

        # A. SET PARAMETERS & RUN
        set_property generic "DISTRIBUTION_TYPE=$dist BURST_PERCENT=$burst" [get_filesets sim_1]

        reset_simulation -simset sim_1 -mode behavioral
        launch_simulation
        run -all
        close_sim -force

        # B. RETRIEVE DATA IMMEDIATELY
        set generated_files [glob -nocomplain ./*.sim/sim_1/behav/xsim/$verilog_filename]

        if {$generated_files == ""} {
            set generated_files [glob -nocomplain ./*/*.sim/sim_1/behav/xsim/$verilog_filename]
        }

        if {$generated_files != ""} {

            set source_file [lindex $generated_files 0]

            # Read the content generated by Verilog
            if {![catch {exec cat $source_file} content]} {

                # Append to the file corresponding to this distribution
                set fout [open $final_output "a"]
                puts $fout $content
                close $fout

                puts "  -> Saved to $final_output"

            } else {
                puts "  ERROR: Could not read $source_file"
            }

        } else {
            puts "  ERROR: Verilog output file not found for this run."
        }
    }
}

puts "----------------------------------------------------------------"
puts "SWEEP COMPLETE."
puts "Generated:"
puts "  FINAL_ALL_RESULTS_UNIFORM.txt"
puts "  FINAL_ALL_RESULTS_ZIPF.txt"
puts "  FINAL_ALL_RESULTS_SKEWED.txt"
puts "----------------------------------------------------------------"
